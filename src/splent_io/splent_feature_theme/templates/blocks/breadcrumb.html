{# Composable breadcrumb block. The separators are drawn by CSS: a "/" written here would be read out between every pair of labels. #} {%- if breadcrumb %}

    2. {%- elif crumb.url %}
  2. {{ crumb.label }}
    3. {%- else %} {#- A level the reader cannot open, such as a grouping with no page of its own. It is still part of the trail. #}
  3. {{ crumb.label }}
    4. {%- endif %} {%- endfor %}
{%- endif %}
